% Dual-mode timing diagram: pull and push interaction over a 6-week task
% Shows how push mode keeps dormant-task memory current
\begin{figure}[t]
\centering
\begin{tikzpicture}[
    >=stealth,
    xscale=1.8,
    yscale=0.9,
    event/.style={circle, fill=#1, inner sep=1.5pt},
    pullev/.style={event=MidnightBlue},
    pushev/.style={event=BrickRed},
    commitdot/.style={event=gray!60, inner sep=1pt},
    brace/.style={decorate, decoration={brace, amplitude=4pt, mirror}},
  ]

  % === Time axis ===
  \draw[->, thick] (0,0) -- (7.2,0) node[right, font=\small] {weeks};
  \foreach \w in {1,...,6} {
    \draw (\w, 0.08) -- (\w, -0.08) node[below, font=\scriptsize] {\w};
  }

  % === Task A timeline (top) ===
  \node[font=\small\bfseries, anchor=east] at (-0.15, 2.8) {Task A};
  \draw[thick, gray!40] (0.3, 2.8) -- (6.5, 2.8);

  % Active periods for Task A
  \fill[blue!10, rounded corners=2pt] (0.3, 2.4) rectangle (1.8, 3.2);
  \fill[blue!10, rounded corners=2pt] (5.2, 2.4) rectangle (6.5, 3.2);
  \node[font=\scriptsize, text=MidnightBlue] at (1.05, 3.4) {active};
  \node[font=\scriptsize, text=gray] at (3.5, 3.4) {dormant};
  \node[font=\scriptsize, text=MidnightBlue] at (5.85, 3.4) {resumes};

  % Pull events on Task A
  \node[pullev] at (0.5, 2.8) {};
  \node[pullev] at (0.9, 2.8) {};
  \node[pullev] at (1.3, 2.8) {};
  \node[pullev] at (1.6, 2.8) {};
  \node[pullev] at (5.3, 2.8) {};
  \node[pullev] at (5.8, 2.8) {};
  \node[pullev] at (6.3, 2.8) {};

  % Push events on Task A (during dormant period)
  \node[pushev] at (2.5, 2.8) {};
  \node[pushev] at (3.5, 2.8) {};
  \node[pushev] at (4.5, 2.8) {};

  % === Task B timeline (middle) ===
  \node[font=\small\bfseries, anchor=east] at (-0.15, 1.2) {Task B};
  \draw[thick, gray!40] (1.5, 1.2) -- (5.0, 1.2);

  % Active period for Task B
  \fill[blue!10, rounded corners=2pt] (1.5, 0.8) rectangle (5.0, 1.6);
  \node[font=\scriptsize, text=MidnightBlue] at (3.25, 1.8) {active};

  % Pull events on Task B
  \node[pullev] at (1.7, 1.2) {};
  \node[pullev] at (2.2, 1.2) {};
  \node[pullev] at (2.8, 1.2) {};
  \node[pullev] at (3.3, 1.2) {};
  \node[pullev] at (3.9, 1.2) {};
  \node[pullev] at (4.5, 1.2) {};

  % === Commit activity (bottom, gray) ===
  \node[font=\small\bfseries, anchor=east] at (-0.15, -0.6) {\color{gray}Commits};
  \foreach \x in {0.4, 0.7, 0.8, 1.1, 1.4, 1.6, 1.8, 2.0, 2.3, 2.5, 2.7, 3.0,
                   3.2, 3.4, 3.6, 3.8, 4.0, 4.2, 4.4, 4.7, 5.0, 5.3, 5.5, 5.8,
                   6.0, 6.2, 6.4} {
    \node[commitdot] at (\x, -0.6) {};
  }

  % === Annotations ===
  % Arrow showing push keeps A current
  \draw[BrickRed, ->, thick, dashed]
    (3.5, 2.4) -- (3.5, 1.8)
    node[midway, right, font=\scriptsize, text=BrickRed, xshift=2pt]
    {B's decisions};
  \node[font=\scriptsize, text=BrickRed, align=center] at (3.5, 2.1)
    {propagated\\[-2pt]to A's digest};

  % Brace showing dormant gap
  \draw[brace, thick, gray] (1.9, 2.3) -- (5.1, 2.3)
    node[midway, below=5pt, font=\scriptsize, text=gray, align=center]
    {3.2 weeks dormant\\push keeps digest current};

  % Callout for the key moment: resume with fresh context
  \draw[->, thick, OliveGreen] (5.3, 3.6) -- (5.3, 3.1);
  \node[font=\scriptsize, text=OliveGreen, align=center, anchor=south] at (5.3, 3.6)
    {pull returns digest\\already refreshed\\by push};

  % === Legend ===
  \node[anchor=north west, font=\scriptsize] at (0, -1.3) {
    \tikz\node[pullev] {}; pull (session start) \quad
    \tikz\node[pushev] {}; push (scheduled refresh) \quad
    \tikz\node[commitdot] {}; repository commits
  };

\end{tikzpicture}
\caption{Dual-mode retrieval over a six-week horizon with two concurrent tasks.
  Task~A is active in weeks~1--2, then dormant while Task~B runs in weeks~2--5.
  During dormancy, push-mode refresh propagates decisions from Task~B into Task~A's
  memory digest at a frequency proportional to commit activity.
  When Task~A resumes in week~5, the pull query returns context that is already current ---
  no expensive re-derivation required.
  Without push mode, the week-5 pull would return a stale digest from week~2, missing
  three weeks of relevant architectural decisions.}
\label{fig:dual-mode}
\end{figure}
